cmake_minimum_required(VERSION 3.20)
project(vcortex_llama_ring LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.deps/llama.cpp" CACHE PATH "Pinned, patched llama.cpp source")
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
  message(FATAL_ERROR "Run python3 scripts/bootstrap.py first")
endif()
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO ON CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_executable(llama-ring src/main.cpp src/runtime.cpp src/self_test.cpp src/hybrid_test.cpp)
target_include_directories(llama-ring PRIVATE "${LLAMA_SOURCE}/vendor")
target_link_libraries(llama-ring PRIVATE llama ggml)
target_compile_options(llama-ring PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(llama-ring PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
enable_testing()
add_executable(llama-stop-test src/stop_test.cpp)
add_test(NAME streaming-stop COMMAND llama-stop-test)
add_test(NAME cli-help COMMAND llama-ring --help)
add_executable(llama-shadow src/shadow_main.cpp src/shadow.cpp src/shadow_test.cpp src/ram_test.cpp src/runtime.cpp)
target_include_directories(llama-shadow PRIVATE "${LLAMA_SOURCE}/vendor")
target_link_libraries(llama-shadow PRIVATE llama ggml)
target_compile_options(llama-shadow PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(llama-shadow PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_test(NAME shadow-cli-help COMMAND llama-shadow --help)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(llama-shadow-worker src/server_worker.cpp src/vision.cpp src/mtp.cpp src/checkpoint.cpp src/shadow.cpp src/runtime.cpp)
target_include_directories(llama-shadow-worker PRIVATE "${LLAMA_SOURCE}/vendor" "${LLAMA_SOURCE}/common" "${LLAMA_SOURCE}/src")
target_link_libraries(llama-shadow-worker PRIVATE llama-common llama ggml mtmd OpenSSL::Crypto Threads::Threads)
target_compile_options(llama-shadow-worker PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(llama-shadow-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_test(NAME worker-cli-help COMMAND llama-shadow-worker --help)
option(RING_BUILD_SPEED_PROBE "Build the paired runtime speed probe" OFF)
if(RING_BUILD_SPEED_PROBE)
  add_executable(ring-speed src/speed_probe.cpp src/runtime.cpp)
  target_compile_definitions(ring-speed PRIVATE RING_SPEED_RUNTIME=1)
  target_include_directories(ring-speed PRIVATE "${LLAMA_SOURCE}/vendor")
  target_link_libraries(ring-speed PRIVATE llama ggml)
  target_compile_options(ring-speed PRIVATE -Wall -Wextra -Wpedantic)
  set_target_properties(ring-speed PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()

# Offline benchmark only; shares the production runtime without changing the worker.
add_executable(llama-memory-bench src/memory_bench.cpp src/shadow.cpp src/runtime.cpp)
target_include_directories(llama-memory-bench PRIVATE "${LLAMA_SOURCE}/vendor")
target_link_libraries(llama-memory-bench PRIVATE llama ggml)
target_compile_options(llama-memory-bench PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(llama-memory-bench PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_test(NAME memory-bench-help COMMAND llama-memory-bench --help)

target_link_libraries(llama-session-test PRIVATE llama ggml)
set_target_properties(llama-session-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

add_executable(llama-mtp-test src/mtp_test.cpp src/mtp.cpp src/checkpoint.cpp src/shadow.cpp src/runtime.cpp)
target_include_directories(llama-mtp-test PRIVATE "${LLAMA_SOURCE}/vendor" "${LLAMA_SOURCE}/common" "${LLAMA_SOURCE}/src")
target_link_libraries(llama-mtp-test PRIVATE llama-common llama ggml Threads::Threads)

target_include_directories(llama-qsa-mtp-bench PRIVATE "${LLAMA_SOURCE}/vendor" "${LLAMA_SOURCE}/common" "${LLAMA_SOURCE}/src")
